```typescript
/**
 * Build a NotePositionMap by querying StaveNote absolute positions after draw.
 * Must be called after all stave notes have been drawn.
 *
 * @param staveNotes - array of rendered StaveNote instances (in timeline order)
 * @param staveY     - top y coordinate of the stave these notes belong to
 * @param measureIndex - 0-based measure
 * @param voiceIndex   - 0-based voice
 * @param startNoteIndex - cumulative note index for this batch (for global noteIndex key)
 * @param map        - target map to write into
 */
function appendNotePositions(
  staveNotes: import('vexflow').StaveNote[],
  staveY: number,
  measureIndex: number,
  voiceIndex: number,
  startNoteIndex: number,
  map: import('./types.js').NotePositionMap
): void {
  for (let i = 0; i < staveNotes.length; i++) {
    const sn = staveNotes[i]
    let x = 0
    let width = 12
    let svgEl: Element | null = null
    try {
      x = sn.getAbsoluteX()
      // getBoundingClientRect not available in SVG context; use getBounds if available
      const bounds = (sn as unknown as { getBounds?: () => { w: number } }).getBounds?.()
      if (bounds) width = bounds.w
      svgEl = (sn as unknown as { attrs?: { el?: Element } }).attrs?.el ?? null
    } catch {
      // ignore — position unavailable (e.g. note not yet drawn)
    }
    map.set(startNoteIndex + i, {
      x,
      y: staveY,
      width,
      measureIndex,
      voiceIndex,
      svgElement: svgEl,
    })
  }
}
```

### What this task does

A `Cursor` is a thin SVG overlay (a `<line>` or `<rect>`) drawn on top of the rendered score. It exposes `moveTo(x, y)` (animate or jump to position), `attach(svgContainer)` to insert the overlay, and `detach()` to remove it. /**
```typescript
 * Cursor — SVG overlay that tracks the currently playing note position.
 *
 * Browser-only. Guard all DOM access behind `typeof document !== 'undefined'`.
 */

export interface CursorOptions {
  /** CSS color string. Default '#1565c0'. */
  color?: string
  /** 'line' = vertical line; 'highlight-box' = translucent rect behind note head. Default 'line'. */
  style?: 'line' | 'highlight-box'
  /** Whether to use CSS transition for smooth movement. Default true. */
  animate?: boolean
  /** Height of the cursor line / box in pixels. Default 80. */
  height?: number
}

const NS = 'http://www.w3.org/2000/svg'

export class Cursor {
  private readonly color: string
  private readonly style: 'line' | 'highlight-box'
  private readonly animate: boolean
  private readonly height: number
  private el: SVGElement | null = null
  private container: SVGSVGElement | null = null

  constructor(options: CursorOptions = {}) {
    this.color = options.color ?? '#1565c0'
    this.style = options.style ?? 'line'
    this.animate = options.animate ?? true
    this.height = options.height ?? 80
  }

  /**
   * Insert the cursor element into an SVG container element.
   * Idempotent — calling attach() on an already-attached cursor first detaches.
   */
  attach(svgContainer: SVGSVGElement): void {
    this.detach()
    this.container = svgContainer
    this.el = this.createElement()
    svgContainer.appendChild(this.el)
  }

  /**
   * Remove the cursor element from its container.
   */
  detach(): void {
    if (this.el && this.el.parentNode) {
      this.el.parentNode.removeChild(this.el)
    }
    this.el = null
    this.container = null
  }

  /**
   * Move the cursor to the given x position (note head x) and y position (stave top).
   * No-op if not attached.
   */
  moveTo(x: number, y: number): void {
    if (!this.el) return
    if (this.style === 'line') {
      const line = this.el as SVGLineElement
      line.setAttribute('x1', String(x))
      line.setAttribute('x2', String(x))
      line.setAttribute('y1', String(y))
      line.setAttribute('y2', String(y + this.height))
    } else {
      const rect = this.el as SVGRectElement
      rect.setAttribute('x', String(x - 6))
      rect.setAttribute('y', String(y))
      rect.setAttribute('width', '24')
      rect.setAttribute('height', String(this.height))
    }
  }

  // ─── Private ─────────────────────────────────────────────────

  private createElement(): SVGElement {
    if (this.style === 'line') {
      const line = document.createElementNS(NS, 'line') as SVGLineElement
      line.setAttribute('data-maestro-cursor', 'true')
      line.setAttribute('stroke', this.color)
      line.setAttribute('stroke-width', '2')
      line.setAttribute('x1', '0')
      line.setAttribute('x2', '0')
      line.setAttribute('y1', '0')
      line.setAttribute('y2', String(this.height))
      line.setAttribute('pointer-events', 'none')
      if (this.animate) {
        line.style.transition = 'x1 0.08s linear, x2 0.08s linear'
      }
      return line
    } else {
      const rect = document.createElementNS(NS, 'rect') as SVGRectElement
      rect.setAttribute('data-maestro-cursor', 'true')
      rect.setAttribute('fill', this.color)
      rect.setAttribute('rx', '3')
      rect.setAttribute('x', '0')
      rect.setAttribute('y', '0')
      rect.setAttribute('width', '24')
      rect.setAttribute('height', String(this.height))
      rect.setAttribute('pointer-events', 'none')
      if (this.animate) {
        rect.style.transition = 'x 0.08s linear'
      }
      return rect
    }
  }
}
```

```typescript
render(target: string | HTMLElement, options?: RenderOptions): this {
  const element =
    typeof target === 'string'
      ? typeof document !== 'undefined'
        ? document.querySelector(target)
        : null
      : target

  if (!element) {
    throw new Error(`Render target "${target}" not found.`)
  }

  this.notePositionMap = VexFlowAdapter.render(
    this.score,
    element as HTMLElement,
    this.toRendererOptions(options)
  )

  // Find the last SVG element appended (used by cursor attachment)
  if (typeof document !== 'undefined') {
    const svgs = (element as HTMLElement).querySelectorAll('svg')
    this.renderedSvgContainer = svgs.length > 0
      ? (svgs[svgs.length - 1] as SVGSVGElement)
      : null
  }

  return this
}
```

Add the `enableCursor` and `disableCursor` methods:

```typescript
/**
 * Enable the playback cursor.
 * Attaches an SVG overlay to the rendered score container.
 * Call render() first so the SVG container exists.
 * Browser-only — no-op in Node.
 */
enableCursor(options?: CursorOptions): this {
  if (typeof document === 'undefined') return this

  // Detach any existing cursor
  this.cursor?.detach()

  this.cursor = new Cursor(options)

  if (this.renderedSvgContainer) {
    this.cursor.attach(this.renderedSvgContainer)
  }

  // Wire cursor movement to timing callbacks
  const mergedTimeline = Scheduler.mergeTies(Scheduler.buildTimeline(this.score))
  this.cursorTimingCallbacks = new TimingCallbacks(mergedTimeline, {
    onNote: (event) => {
      const pos = this.notePositionMap.get(event.noteIndex)
      if (pos && this.cursor) {
        this.cursor.moveTo(pos.x, pos.y)
      }
    },
  })

  // If audio adapter is playing, wire callbacks into it
  if (this.audioAdapter) {
    this.audioAdapter.setTimingCallbacks({
      onNote: (event) => {
        const pos = this.notePositionMap.get(event.noteIndex)
        if (pos && this.cursor) {
          this.cursor.moveTo(pos.x, pos.y)
        }
      },
    })
  }

  return this
}

/**
 * Remove the playback cursor overlay.
 */
disableCursor(): this {
  this.cursor?.detach()
  this.cursor = null
  this.cursorTimingCallbacks = null
  return this
}
```

```typescript
private setupClickDelegation(container: HTMLElement): void {
  this.clickListenerCleanup?.()

  const listener = (e: Event) => {
    const target = e.target as Element | null
    if (!target) return

    let el: Element | null = target
    while (el && el !== container) {
      const noteIndexAttr = el.getAttribute('data-note-index')
      const measureIndexAttr = el.getAttribute('data-measure-index')
      if (noteIndexAttr !== null && measureIndexAttr !== null) {
        const noteIndex = parseInt(noteIndexAttr, 10)
        const measureIndex = parseInt(measureIndexAttr, 10)
        this.handleNoteClick(noteIndex, measureIndex, el)
        return
      }
      el = el.parentElement
    }
  }

  container.addEventListener('click', listener)
  this.clickListenerCleanup = () => container.removeEventListener('click', listener)
}

private handleNoteClick(noteIndex: number, measureIndex: number, svgEl: Element): void {
  if (this.selectedNote) {
    const prevEl = this.notePositionMap.get(this.selectedNote.noteIndex)?.svgElement
    prevEl?.classList.remove('maestro-note-selected')
  }

  const pos = this.notePositionMap.get(noteIndex)
  if (!pos) return

  const parts = this.score.getParts()
  const part = parts[pos.voiceIndex]
  if (!part) return
  const voice = part.getVoices()[0]
  if (!voice) return
  const measures = voice.getMeasures()
  const measure = measures[measureIndex]
  if (!measure) return
  const notes = measure.getNotes()

  // Map flat timeline noteIndex to a local index within this measure
  const localNoteIndex = noteIndex - (pos.measureIndex > 0
    ? measures.slice(0, pos.measureIndex).reduce((sum, m) => sum + m.getNotes().length, 0)
    : 0)
  const note = notes[localNoteIndex]
  if (!note) return

  const info: NoteClickInfo = {
    measureIndex,
    voiceIndex: pos.voiceIndex,
    noteIndex,
    note,
  }

  this.selectedNote = info
  svgEl.classList.add('maestro-note-selected')

  for (const handler of this.clickHandlers) {
    try { handler(info) } catch { /* swallow handler errors */ }
  }
}
```

```typescript
import type { Score } from '../../model/Score.js'
import { Scheduler } from '../../scheduler/Scheduler.js'
import type { Timeline } from '../../scheduler/timeline.js'
import type { PlaybackEventType, PlaybackEventHandler } from './ToneAdapter.js'

export interface SoundfontOptions {
  instrument?: string
  /** Soundfont library name. Default 'MusyngKite'. */
  soundfont?: 'MusyngKite' | 'FluidR3_GM'
  /** Base CDN URL for soundfont files. Defaults to jsDelivr + gleitz/midi-js-soundfonts. */
  soundfontUrl?: string
  voices?: string[]
}

const DEFAULT_SOUNDFONT_URL =
  'https://cdn.jsdelivr.net/gh/gleitz/midi-js-soundfonts@gh-pages'

const MIDI_NOTE_NAMES = [
  'C', 'Db', 'D', 'Eb', 'E', 'F', 'Gb', 'G', 'Ab', 'A', 'Bb', 'B',
]

/**
 * Audio adapter that plays back instrument samples loaded from a CDN soundfont.
 * Uses the Web Audio API directly (AudioContext + AudioBufferSourceNode).
 *
 * Implements the same interface surface as ToneAdapter so Song can use either.
 */
export class SoundfontAdapter {
  private audioContext: AudioContext | null = null
  private buffers: Map<number, AudioBuffer> = new Map() // midi note number => AudioBuffer
  private activeSources: AudioBufferSourceNode[] = []
  private score: Score | null = null
  private timeline: Timeline = []
  private eventHandlers: Map<PlaybackEventType, PlaybackEventHandler[]> = new Map()
  private _isPlaying = false
  private _isPaused = false
  private pauseTime = 0
  private startTime = 0
  private instrumentName = 'acoustic_grand_piano'
  private soundfontName: 'MusyngKite' | 'FluidR3_GM' = 'MusyngKite'
  private soundfontBaseUrl = DEFAULT_SOUNDFONT_URL
  private scheduledTimeouts: ReturnType<typeof setTimeout>[] = []

  /**
   * Convert a MIDI note number to a note name (e.g. 60 => 'C4').
   */
  static midiToNoteName(midi: number): string {
    const octave = Math.floor(midi / 12) - 1
    const noteIndex = midi % 12
    return `${MIDI_NOTE_NAMES[noteIndex]}${octave}`
  }

  /**
   * Build the CDN URL for a specific instrument's soundfont file.
   */
  static getSoundfontUrl(
    instrument: string,
    soundfont: 'MusyngKite' | 'FluidR3_GM' = 'MusyngKite',
    baseUrl: string = DEFAULT_SOUNDFONT_URL
  ): string {
    return `${baseUrl}/${soundfont}/${instrument}-mp3.js`
  }

  /**
   * Load a score and configure the instrument. Sample fetching is deferred to play().
   */
  load(score: Score, options?: SoundfontOptions): void {
    this.dispose()
    this.score = score
    this.instrumentName = gmInstrumentName(options?.instrument ?? 'piano')
    this.soundfontName = options?.soundfont ?? 'MusyngKite'
    this.soundfontBaseUrl = options?.soundfontUrl ?? DEFAULT_SOUNDFONT_URL
    this.timeline = Scheduler.mergeTies(Scheduler.buildTimeline(score))
  }

  /**
   * Start playback. Loads samples on first call.
   * Browser-only — rejects in Node (AudioContext not available).
   */
  async play(): Promise<void> {
    if (typeof AudioContext === 'undefined') {
      throw new Error('SoundfontAdapter requires a browser AudioContext.')
    }
    if (this._isPlaying && !this._isPaused) return
    if (!this.score) throw new Error('No score loaded. Call load() before play().')

    if (!this.audioContext) {
      this.audioContext = new AudioContext()
    }
    if (this.audioContext.state === 'suspended') {
      await this.audioContext.resume()
    }

    if (this.buffers.size === 0) {
      await this.loadSamples()
    }

    if (this._isPaused) {
      this.scheduleTimeline(this.pauseTime)
      this._isPaused = false
      this._isPlaying = true
      return
    }

    this.scheduleTimeline(0)
    this._isPlaying = true
    this._isPaused = false
  }

  /** Pause playback (resumable). */
  pause(): void {
    if (!this._isPlaying) return
    this.pauseTime =
      this.audioContext ? this.audioContext.currentTime - this.startTime : 0
    this.clearScheduled()
    for (const src of this.activeSources) {
      try { src.stop() } catch { /* already stopped */ }
    }
    this.activeSources = []
    this._isPaused = true
  }

  /** Stop playback and reset position. */
  stop(): void {
    this.clearScheduled()
    for (const src of this.activeSources) {
      try { src.stop() } catch { /* already stopped */ }
    }
    this.activeSources = []
    this.pauseTime = 0
    this._isPlaying = false
    this._isPaused = false
  }

  /** Register an event handler. */
  on(event: PlaybackEventType, handler: PlaybackEventHandler): void {
    const handlers = this.eventHandlers.get(event) ?? []
    handlers.push(handler)
    this.eventHandlers.set(event, handlers)
  }

  /** Remove an event handler. */
  off(event: PlaybackEventType, handler: PlaybackEventHandler): void {
    const handlers = this.eventHandlers.get(event)
    if (handlers) {
      const idx = handlers.indexOf(handler)
      if (idx !== -1) handlers.splice(idx, 1)
    }
  }

  /** Release all resources. */
  dispose(): void {
    this.stop()
    this.buffers.clear()
    this.timeline = []
    this.score = null
    if (this.audioContext) {
      this.audioContext.close().catch(() => {})
      this.audioContext = null
    }
  }

  get isPlaying(): boolean { return this._isPlaying }
  get isPaused(): boolean { return this._isPaused }

  // ─── Private ─────────────────────────────────────────────────

  private async loadSamples(): Promise<void> {
    if (!this.audioContext) return

    const midiNotes = new Set<number>()
    for (const ev of this.timeline) {
      if (ev.note.midi !== null) midiNotes.add(ev.note.midi)
    }

    const baseUrl = `${this.soundfontBaseUrl}/${this.soundfontName}/${this.instrumentName}-mp3`

    // Load samples in chunks to avoid flooding the CDN
    const notes = [...midiNotes]
    const chunkSize = 16
    for (let i = 0; i < notes.length; i += chunkSize) {
      const chunk = notes.slice(i, i + chunkSize)
      await Promise.all(
        chunk.map(async (midi) => {
          const noteName = SoundfontAdapter.midiToNoteName(midi)
          const url = `${baseUrl}/${noteName}.mp3`
          try {
            const response = await fetch(url)
            if (!response.ok) return
            const arrayBuffer = await response.arrayBuffer()
            const audioBuffer = await this.audioContext!.decodeAudioData(arrayBuffer)
            this.buffers.set(midi, audioBuffer)
          } catch {
            // Sample unavailable — note will play silently
          }
        })
      )
    }
  }

  private scheduleTimeline(offsetSeconds: number): void {
    if (!this.audioContext) return
    this.clearScheduled()
    this.startTime = this.audioContext.currentTime - offsetSeconds
    let lastMeasure = -1

    for (const ev of this.timeline) {
      if (ev.time < offsetSeconds) continue
      const audioTime = this.startTime + ev.time
      const delay = (audioTime - this.audioContext.currentTime) * 1000

      const id = setTimeout(() => {
        this.playSample(ev.note.midi, ev.note.duration, ev.note.velocity)
        this.emitEvent('note', {
          pitch: ev.note.pitch,
          midi: ev.note.midi,
          duration: ev.note.duration,
          time: ev.time,
          measure: ev.note.measure,
          beat: ev.note.beat,
        })
        if (ev.note.measure !== lastMeasure) {
          lastMeasure = ev.note.measure
          this.emitEvent('measure', { measure: ev.note.measure, time: ev.time })
        }
      }, Math.max(0, delay))

      this.scheduledTimeouts.push(id)
    }

    if (this.timeline.length > 0) {
      const lastEv = this.timeline[this.timeline.length - 1]
      const endDelay =
        (this.startTime + lastEv.time + lastEv.note.duration - this.audioContext.currentTime) * 1000
      const endId = setTimeout(() => {
        this.emitEvent('end', {})
        this._isPlaying = false
      }, Math.max(0, endDelay + 50))
      this.scheduledTimeouts.push(endId)
    }
  }

  private playSample(midi: number | null, duration: number, velocity: number): void {
    if (!this.audioContext || midi === null) return
    const buffer = this.buffers.get(midi)
    if (!buffer) return

    const source = this.audioContext.createBufferSource()
    source.buffer = buffer

    const gainNode = this.audioContext.createGain()
    gainNode.gain.value = velocity / 127

    source.connect(gainNode)
    gainNode.connect(this.audioContext.destination)

    source.start(0)
    source.stop(this.audioContext.currentTime + duration)

    this.activeSources.push(source)
    source.onended = () => {
      const idx = this.activeSources.indexOf(source)
      if (idx !== -1) this.activeSources.splice(idx, 1)
    }
  }

  private clearScheduled(): void {
    for (const id of this.scheduledTimeouts) clearTimeout(id)
    this.scheduledTimeouts = []
  }

  private emitEvent(type: PlaybackEventType, data: Record<string, unknown>): void {
    const handlers = this.eventHandlers.get(type) ?? []
    for (const handler of handlers) {
      try { handler(data) } catch { /* swallow handler errors */ }
    }
  }
}

/**
 * Map maestro instrument names to GM soundfont file names.
 */
function gmInstrumentName(name: string): string {
  const map: Record<string, string> = {
    piano: 'acoustic_grand_piano',
    strings: 'string_ensemble_1',
    choir: 'choir_aahs',
    organ: 'church_organ',
    violin: 'violin',
    guitar: 'acoustic_guitar_nylon',
    flute: 'flute',
    trumpet: 'trumpet',
    synth: 'synth_lead_1_square',
  }
  return map[name] ?? name
}
```

```typescript
/**
 * Set the zoom scale factor for rendering. 1.0 = 100%, 1.5 = 150%.
 * If render() has already been called, immediately re-renders at the new scale.
 */
setZoom(factor: number): this {
  this.zoomFactor = factor
  if (this.renderTarget) {
    this.rerender()
  }
  return this
}

/**
 * Enable or disable responsive re-render when the container is resized.
 * Requires render() to have been called first.
 * Browser-only — no-op in Node.
 */
setResponsive(enabled: boolean): this {
  if (typeof ResizeObserver === 'undefined') return this

  if (this.resizeObserver) {
    this.resizeObserver.disconnect()
    this.resizeObserver = null
  }

  if (enabled && this.renderTarget) {
    this.resizeObserver = new ResizeObserver((entries) => {
      for (const entry of entries) {
        const newWidth = Math.round(entry.contentRect.width)
        if (newWidth > 0) {
          this.lastRenderOptions = { ...this.lastRenderOptions, width: newWidth }
          this.rerender()
        }
      }
    })
    this.resizeObserver.observe(this.renderTarget)
  }

  return this
}

/** Re-render into the saved target with current options and zoom. */
private rerender(): void {
  if (!this.renderTarget) return
  // Remove all child nodes from the target before re-rendering
  while (this.renderTarget.firstChild) {
    this.renderTarget.removeChild(this.renderTarget.firstChild)
  }
  this.notePositionMap = VexFlowAdapter.render(
    this.score,
    this.renderTarget,
    this.toRendererOptions({ ...this.lastRenderOptions, zoom: this.zoomFactor })
  )
  // Re-attach cursor overlay if active
  if (this.cursor) {
    const svgs = this.renderTarget.querySelectorAll('svg')
    this.renderedSvgContainer = svgs.length > 0
      ? (svgs[svgs.length - 1] as SVGSVGElement)
      : null
    if (this.renderedSvgContainer) {
      this.cursor.attach(this.renderedSvgContainer)
    }
  }
}
```
